Exercise 1: Sum of List (For Loop + Function)

📌 Write a function sum_list(numbers) that takes a list of numbers and returns their sum using a for loop.

numbers = [1, 2, 3, 4, 5]
print(sum_list(numbers))  # Output: 15

def sum_list(numbers):
    total = 0
    for num in numbers:
        total += num
    return total

numbers = [1, 2, 3, 4, 5]
print(sum_list(numbers))

Additional Example: Sum of Even Numbers

numbers = [1, 2, 3, 4, 5, 6]
print(sum_evens(numbers))  # Output: 12

def sum_evens(numbers):
    total = 0
    for num in numbers:
        if num % 2 == 0:
            total += num
    return total

numbers = [1, 2, 3, 4, 5, 6]
print(sum_evens(numbers))

def count_evens(numbers):
    count = 0

Exercise 3: Reverse a List (For Loop + Function)

📌 Write a function reverse_list(lst) that reverses a list without using [::-1] or .reverse().

my_list = [10, 20, 30, 40]
print(reverse_list(my_list))  # Output: [40, 30, 20, 10]

def reverse_list(lst):
    reversed_lst = []
    for item in lst:
        reversed_lst.insert(0, item)
    return reversed_lst

my_list = [10, 20, 30, 40]
print(reverse_list(my_list))

Additional Example: Reverse Strings in List

words = ["hello", "world", "python"]
print(reverse_strings(words))  # Output: ['olleh', 'dlrow', 'nohtyp']

def reverse_strings(lst):
    reversed_lst = []
    for item in lst:
        reversed_lst.append(item[::-1])
    return reversed_lst

words = ["hello", "world", "python"]
print(reverse_strings(words))

def find_max(numbers):
    max_num = numbers[0]

    return max_num

Exercise 5: Remove Duplicates (For Loop + Function)

📌 Write a function remove_duplicates(lst) that removes duplicate items from a list and returns a new list.

my_list = [1, 2, 2, 3, 4, 4, 5]
print(remove_duplicates(my_list))  # Output: [1, 2, 3, 4, 5]

def remove_duplicates(lst):
    new_list = []
    for item in lst:
        if item not in new_list:
            new_list.append(item)
    return new_list

my_list = [1, 2, 2, 3, 4, 4, 5]
print(remove_duplicates(my_list))

Additional Example: Keep First Occurrence

my_list = ["apple", "banana", "apple", "cherry"]
print(keep_first(my_list))  # Output: ['apple', 'banana', 'cherry']

def keep_first(lst):
    seen = []
    for item in lst:
        if item not in seen:
            seen.append(item)
    return seen

my_list = ["apple", "banana", "apple", "cherry"]
print(keep_first(my_list))

Exercise 6: Fibonacci Sequence (While Loop + Function)

📌 Write a function fibonacci(n) that generates the first n Fibonacci numbers using a while loop.

print(fibonacci(7))  # Output: [0, 1, 1, 2, 3, 5, 8]

def fibonacci(n):
    fib = [0, 1]
    while len(fib) < n:
        fib.append(fib[-1] + fib[-2])
    return fib[:n]

print(fibonacci(7))

Additional Example: Fibonacci Up to Value

print(fibonacci_up_to(10))  # Output: [0, 1, 1, 2, 3, 5, 8]

def fibonacci_up_to(max_value):
    fib = [0, 1]
    while fib[-1] + fib[-2] <= max_value:
        fib.append(fib[-1] + fib[-2])
    return fib

print(fibonacci_up_to(10))

Exercise 8: Prime Checker (For Loop + Function)

📌 Write a function is_prime(num) that checks if a number is prime (returns True or False).

print(is_prime(7))  # Output: True
print(is_prime(4))  # Output: False

def is_prime(num):
    if num <= 1:
        return False
    for i in range(2, int(num ** 0.5) + 1):
        if num % i == 0:
            return False
    return True

print(is_prime(7))
print(is_prime(4))

Additional Example: List of Primes

print(get_primes(10))  # Output: [2, 3, 5, 7]

def is_prime(num):
    if num <= 1:
        return False
    for i in range(2, int(num ** 0.5) + 1):
        if num % i == 0:
            return False
    return True

def get_primes(n):
    primes = []
    for num in range(2, n + 1):
        if is_prime(num):
            primes.append(num)
    return primes

print(get_primes(10))

Exercise 10: Multiplication Table (Nested Loops + Function)

📌 Write a function multiplication_table(n) that prints a multiplication table up to n x n.

multiplication_table(3)
# Output:
# 1 2 3
# 2 4 6
# 3 6 9

def multiplication_table(n):
    for i in range(1, n + 1):
        row = []
        for j in range(1, n + 1):
            row.append(i * j)
        print(" ".join(map(str, row)))

multiplication_table(3)

Additional Example: Addition Table

addition_table(3)
# Output:

def addition_table(n):
    for i in range(1, n + 1):
        row = []
        for j in range(1, n + 1):
            row.append(i + j)
        print(" ".join(map(str, row)))

addition_table(3)
